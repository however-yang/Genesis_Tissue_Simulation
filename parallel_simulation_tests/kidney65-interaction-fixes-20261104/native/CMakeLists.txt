cmake_minimum_required(VERSION 3.18)
project(EndoscopicTissueHandoff LANGUAGES CXX CUDA)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CUDA_STANDARD 17)
set(CMAKE_CUDA_STANDARD_REQUIRED ON)
find_package(CUDAToolkit REQUIRED)
add_library(LiverCudaSim SHARED
    vendor/src/plugin_api.cpp
    vendor/cuda/training/state_archive.cpp
    vendor/cuda/dc_recon.cu
    vendor/cuda/physics.cu
    vendor/cuda/cut.cu
    vendor/cuda/haptics.cu
    vendor/cuda/xpbd/xpbd_solver.cu)
target_include_directories(LiverCudaSim PRIVATE vendor/cuda)
target_link_libraries(LiverCudaSim PRIVATE CUDA::cudart_static)
if(ENDO_KIDNEY_CLOSED_STATS)
    target_compile_definitions(LiverCudaSim PRIVATE ENDO_KIDNEY_CLOSED_STATS)
endif()
if(ENDO_KIDNEY_CLOSED_RAY)
    target_compile_definitions(LiverCudaSim PRIVATE ENDO_KIDNEY_CLOSED_RAY)
endif()
if(ENDO_KIDNEY_CLOSED_RAY_PROBE)
    if(NOT ENDO_FRAGMENT_PROBE)
        message(FATAL_ERROR "Closed ray is probe-only until live closure certification is implemented")
    endif()
    target_compile_definitions(LiverCudaSim PRIVATE ENDO_KIDNEY_CLOSED_RAY_PROBE)
endif()
if(ENDO_KIDNEY_SPLIT_FRAGMENT)
    target_compile_definitions(LiverCudaSim PRIVATE ENDO_KIDNEY_SPLIT_FRAGMENT)
endif()
if(ENDO_KIDNEY_CACHE_TREE_BOUNDS)
    target_compile_definitions(LiverCudaSim PRIVATE ENDO_KIDNEY_CACHE_TREE_BOUNDS)
endif()
if(ENDO_KIDNEY_MG_WARP_REDUCE)
    if(NOT ENDO_KIDNEY_PRECISE_MG)
        message(FATAL_ERROR "MG warp reduction requires wide MG accumulators")
    endif()
    target_compile_definitions(LiverCudaSim PRIVATE ENDO_KIDNEY_MG_WARP_REDUCE)
endif()
if(ENDO_KIDNEY_PRECISE_MG)
    target_compile_definitions(LiverCudaSim PRIVATE ENDO_KIDNEY_PRECISE_MG)
endif()
if(ENDO_KIDNEY_PRECISE_COARSE_WARP)
    target_compile_definitions(LiverCudaSim PRIVATE ENDO_KIDNEY_PRECISE_COARSE_WARP)
endif()
if(ENDO_KIDNEY_PRECISE_LINE)
    target_compile_definitions(LiverCudaSim PRIVATE ENDO_KIDNEY_PRECISE_LINE)
endif()
if(ENDO_KIDNEY_STAGE_TRACE)
    target_compile_definitions(LiverCudaSim PRIVATE ENDO_KIDNEY_STAGE_TRACE)
endif()
set(ENDO_FRAGMENT_ANGLE_BUDGET "0.02" CACHE STRING "Per-node attempt threshold for conservative winding bounds")
if(NOT ENDO_FRAGMENT_ANGLE_BUDGET STREQUAL "0.02" AND NOT ENDO_FRAGMENT_ANGLE_BUDGET STREQUAL "0.04")
    message(FATAL_ERROR "Only reviewed winding-bound settings 0.02 and 0.04 are supported")
endif()
target_compile_definitions(LiverCudaSim PRIVATE ENDO_FRAGMENT_ANGLE_BUDGET=${ENDO_FRAGMENT_ANGLE_BUDGET}f)
option(ENDO_KIDNEY_CONTACT_TRACE "Read-only contact witness trace" OFF)
option(ENDO_KIDNEY_PRECISE_COARSE "Experimental wider coarse accumulation" OFF)
option(ENDO_KIDNEY_WARP_STACK "Experimental per-warp fragment traversal stack" OFF)
option(ENDO_KIDNEY_WARP_SCRATCH "Experimental per-warp exit and constraint scratch" OFF)
if(ENDO_KIDNEY_WARP_SCRATCH)
    target_compile_definitions(LiverCudaSim PRIVATE ENDO_KIDNEY_WARP_SCRATCH)
endif()
option(ENDO_FRAGMENT_PROBE "Read-only frozen fragment-query probes" OFF)
option(ENDO_FRAGMENT_DIPOLE "Experimental first-moment bounded winding expansion" OFF)
if(ENDO_FRAGMENT_DIPOLE)
    target_compile_definitions(LiverCudaSim PRIVATE ENDO_FRAGMENT_DIPOLE)
endif()
option(ENDO_KIDNEY_PRECISE_INTERACTION "Experimental wider interaction sums" OFF)
option(ENDO_KIDNEY_PRECISE_TOOL "Experimental wider tool sums" OFF)
if(ENDO_KIDNEY_PRECISE_TOOL)
    target_compile_definitions(LiverCudaSim PRIVATE ENDO_KIDNEY_PRECISE_TOOL)
endif()
if(ENDO_KIDNEY_PRECISE_INTERACTION)
    target_compile_definitions(LiverCudaSim PRIVATE ENDO_KIDNEY_PRECISE_INTERACTION)
endif()
if(ENDO_KIDNEY_WARP_STACK)
    target_compile_definitions(LiverCudaSim PRIVATE ENDO_KIDNEY_WARP_STACK)
endif()
if(ENDO_FRAGMENT_PROBE)
    target_compile_definitions(LiverCudaSim PRIVATE ENDO_FRAGMENT_PROBE)
endif()
if(ENDO_KIDNEY_PRECISE_COARSE)
    target_compile_definitions(LiverCudaSim PRIVATE ENDO_KIDNEY_PRECISE_COARSE)
endif()
if(ENDO_KIDNEY_CONTACT_TRACE)
    target_compile_definitions(LiverCudaSim PRIVATE ENDO_KIDNEY_CONTACT_TRACE)
endif()
if(MSVC)
    target_compile_options(LiverCudaSim PRIVATE
        $<$<COMPILE_LANGUAGE:CXX>:/utf-8>
        $<$<COMPILE_LANGUAGE:CUDA>:-Xcompiler=/utf-8>)
endif()
set_target_properties(LiverCudaSim PROPERTIES
    CUDA_ARCHITECTURES "80"
    CUDA_SEPARABLE_COMPILATION ON
    POSITION_INDEPENDENT_CODE ON
    LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib"
    RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib"
    RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_BINARY_DIR}/lib")
# Historical Python validators append this basename. The alias points to an
# ELF .so on Linux; it is not a Windows DLL and never occupies a default bin/.
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    add_custom_command(TARGET LiverCudaSim POST_BUILD
        COMMAND ${CMAKE_COMMAND} -E create_symlink
            "$<TARGET_FILE_NAME:LiverCudaSim>"
            "$<TARGET_FILE_DIR:LiverCudaSim>/LiverCudaSim.dll")
endif()
